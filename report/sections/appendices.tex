\chapter{Core Research-Table Schema}
\label{app:schema}

The following excerpts reproduce the Prisma models most relevant to the
research instrumentation, abbreviated to their analytically meaningful
fields. The full schema (approximately ninety models) lives in
\code{apps/api/prisma/schema.prisma}.

\begin{lstlisting}[caption={Facial-analysis result tables.}]
model EmotionFrame {              // @@map("emotion_frames") -- OpenFace 3.0
  jobId               String
  sessionId, userId, courseId  String
  frameWallMs         BigInt
  frameIndex          Int
  faceDetected        Boolean
  pHappiness, pSadness, pSurprise, pFear,
  pAnger, pDisgust, pContempt, pNeutral   Float?   // 8-class AffectNet
  dominantEmotion     String?
  dominantProbability Float?
  headPoseYaw, headPosePitch, headPoseRoll Float?  // NEVER populated (NULL)
}

model PyfeatAuResult {           // @@map("pyfeat_au_results") -- Py-Feat
  jobId       String
  frameIndex  Int
  timestamp   Float
  wallTime    DateTime
  au01, au02, au04, au05, au06, au07, au09, au10, au12,
  au14, au15, au17, au20, au23, au24, au25, au26, au28  Float?  // 18 AUs
  faceConf    Float?
  faceBox     Json?
}
\end{lstlisting}

\begin{lstlisting}[caption={Affective-state windows and self-report basis.}]
model AffectiveStateWindow {      // @@map("affective_state_windows")
  sessionId, userId, courseId  String
  configId, configVersion      // which rule set produced this row
  windowStartWallMs, windowEndWallMs  BigInt
  framesInWindow, framesWithFace      Int
  engagement, boredom, confusion, frustration  Float
  dominantState  String
  meanHappiness ... meanNeutral       Float?     // mean-pooled inputs
}

// Self-report is stored on the activity log:
//   ActivityLog.action == EMOTION_SELF_REPORT
//   ActivityLog.metadata -> { emotion: "engaged|bored|confused|frustrated|neutral" }
\end{lstlisting}

\begin{lstlisting}[caption={Interventions, spaced repetition, and text-mining.}]
model LearningIntervention {      // @@map("learning_interventions")
  userId, courseId  String
  type    InterventionType        // PRACTICE_TESTING | INTERROGATIVE_ELABORATION
                                  // | STEPWISE_LEARNING | DISTRIBUTED_PRACTICE
  status  InterventionStatus      // IN_PROGRESS | COMPLETED | ABANDONED
  selectedText  String
  sessionData   Json?             // activity-specific state
}

model SpacedRepetitionCard {      // @@map("spaced_repetition_cards") -- SM-2
  front, back   String
  ease          Float   @default(2.5)
  interval      Int     @default(0)
  repetitions   Int     @default(0)
  nextReviewAt  DateTime
}

model EfDetection {               // @@map("ef_detections")
  messageId, sessionId, studentId  String
  constructKey  String            // one of nine constructs
  label         String            // 'pending'/'error' rows also occur
  confidence    Float?
  severity      Int?              // populated for confusion only
  rationale, warning  String?
  provider, model  String
  promptVersion Int
  latencyMs     Int?
}
\end{lstlisting}

\begin{lstlisting}[caption={Replay snapshots, sync anchor, and coding.}]
model SessionReplaySnapshot {     // @@map("session_replay_snapshots")
  sessionId, userId  String
  pageUrl            String
  html               String?      // null when DOM capture off (study default)
  screenshotDataUrl  String?
  width, height, scrollX, scrollY  Int
  capturedAt         BigInt
  aois               Json?        // [{ region, x, y, width, height }]
  scrollHosts        Json?        // inner-container scroll (added mid-2026)

}

model session_sync_anchors {
  sessionId  String  @unique
  wallClockMs, monotonicMs, serverReceiveMs  BigInt
  timezone, userAgent  String
}

model ReplayAnnotation {          // @@map("replay_annotations")
  sessionId, researcherId  String
  codeId   String?               // -> ReplayCode, SET NULL on code delete
  startMs  BigInt
  endMs    BigInt?               // null => point annotation
  note     String?
}
\end{lstlisting}

\chapter{Default Intervention Prompt Templates}
\label{app:prompts}

Each intervention pairs a teacher-editable \emph{system prompt} with a
hard-coded \emph{user-prompt template} (\cref{sec:interventions-teacher}).
The following are the default system prompts, summarized from
\code{apps/api/src/learning-interventions/prompts/default-prompts.ts}.
Placeholders such as \code{{{mcqCount}}} are substituted at generation
time.

\begin{description}
  \item[Practice testing.] Generate exactly \code{{{mcqCount}}}
  multiple-choice and \code{{{shortAnswerCount}}} short-answer questions
  over the supplied material; each MCQ has exactly four options; short
  answers include two to four expected keywords; questions target content,
  not slide design; output strict JSON.
  \item[Interrogative elaboration.] Suggest three to five insightful
  ``why'' and ``how'' questions that probe cause--effect relationships and
  mechanisms (roughly balanced), plus the key concepts; the accompanying
  answer prompt explains rather than quizzes; output strict JSON.
  \item[Stepwise learning.] Decompose the material into three to six
  progressive steps, each with a comprehension check (question, hint,
  sample answer); the check prompt grades generously on core-concept
  understanding; output strict JSON.
  \item[Distributed practice.] Create exactly \code{{{cardCount}}}
  flashcards, each testing one atomic concept with a concise back; output
  strict JSON.
\end{description}
